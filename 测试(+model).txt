def load_kitti_data(frame_id, data_dir):
    # 确保frame_id是6位零填充字符串
    frame_str = f"{int(frame_id):06d}"

    calib_path = os.path.join(data_dir, 'calib.txt')

    img_path = Path(data_dir) / 'image_2' / f"{frame_str}.png"
    pc_path = Path(data_dir) / 'velodyne' / f"{frame_str}.bin"

    # 详细路径检查
    if not img_path.exists():
        raise FileNotFoundError(f"图像文件缺失: {img_path}")
    if not pc_path.exists():
        raise FileNotFoundError(f"点云文件缺失: {pc_path}")
    #print(f"正在加载帧 {frame_id} 数据，路径验证：")
    print(f"图像路径: {img_path}")
    #print(f"点云路径: {pc_path}")
    #print(f"标定路径: {Path(data_dir) / 'calib.txt'}")

    # 检查文件大小
    if os.path.getsize(img_path) < 1024:
        raise ValueError(f"图像文件过小: {img_path} (可能损坏)")
    if os.path.getsize(pc_path) < 1024:
        raise ValueError(f"点云文件过小: {pc_path} (可能损坏)")
    # 使用更健壮的图像加载方式
    try:
        image = cv2.imread(str(img_path))
        if image is None:
            from PIL import Image
            pil_img = Image.open(str(img_path)).convert('RGB')
            image = np.array(pil_img)[:, :, ::-1]  # RGB转BGR
    except Exception as e:
        print(f"备用加载方式失败: {str(e)}")
        image = np.zeros((375, 1242, 3), dtype=np.uint8)  # 返回空白图像

    # 验证图像尺寸
    h, w = image.shape[:2]
    if h not in [370, 375] or w not in [1224, 1242]:
        print(f"异常图像尺寸 {img_path}: {w}x{h}")

    # 点云加载异常处理
    try:
        pc_data = np.fromfile(pc_path, dtype=np.float32).reshape(-1, 4)
        if pc_data.size == 0:
            raise ValueError("空点云文件")
    except Exception as e:
        print(f"点云加载异常 {pc_path}: {str(e)}")
        pc_data = np.zeros((0, 4), dtype=np.float32)

    return {
        'image': image,
        'pointcloud': pc_data,
        'calib': parse_kitti_calib(Path(data_dir) / 'calib.txt')
    }


def parse_kitti_calib(calib_path):
    calib = {}
    with open(calib_path) as f:
        for line in f:
            if ':' in line:
                key, value = line.split(':', 1)
                values = list(map(float, value.strip().split()))
                if key == 'P2':
                    calib[key] = np.array(values).reshape(3, 4)
                elif key == 'R0_rect':
                    calib[key] = np.array(values).reshape(3, 3)
                elif key == 'Tr_velo_to_cam':
                    calib[key] = np.array(values).reshape(3, 4)
    return calib


def project_pointcloud(points, calib):
    """后的点云投影"""
    points_homo = np.hstack([points[:, :3], np.ones((points.shape[0], 1))])

    Tr_velo_to_cam = np.vstack([calib['Tr_velo_to_cam'], [0, 0, 0, 1]])
    R_rect = np.eye(4)
    R_rect[:3, :3] = calib['R0_rect']

    points_cam = R_rect @ Tr_velo_to_cam @ points_homo.T
    points_cam = points_cam[:3, :]

    P2 = calib['P2']
    # 点：添加缺失的右括号
    points_img = P2 @ np.vstack([points_cam, np.ones((1, points_cam.shape[1]))])
    points_img = (points_img[:2] / points_img[2]).T
    if points.shape[0] == 0:
        return np.zeros((0, 2), dtype=np.float32)
    valid = (points_img[:, 0] >= 0) & (points_img[:, 0] < 1242) & \
            (points_img[:, 1] >= 0) & (points_img[:, 1] < 375)
    return points_img[valid], valid

class ChainComplex:
    def __init__(self, vertices, edges, faces=None):
        self.dim0 = vertices  # [N, 3] 3D坐标
        self.dim1 = edges  # [E, 2] 边连接
        self.dim2 = faces  # [F, 3] 三角面片

    def boundary_matrix(self, dim):
        if dim == 1:
            bm = torch.zeros(len(self.dim0), len(self.dim1))
            for i, (u, v) in enumerate(self.dim1):
                bm[u, i] = 1
                bm[v, i] = -1
            return bm
        elif dim == 2 and self.dim2 is not None:
            bm = torch.zeros(len(self.dim1), len(self.dim2), dtype=torch.float32)
            edge_dict = {(u, v): idx for idx, (u, v) in enumerate(self.dim1)}
            for f_idx, (v0, v1, v2) in enumerate(self.dim2):
                edges = [(v0, v1), (v1, v2), (v2, v0)]
                for u, v in edges:
                    if (u, v) in edge_dict:
                        bm[edge_dict[(u, v)], f_idx] = 1
                    elif (v, u) in edge_dict:
                        bm[edge_dict[(v, u)], f_idx] = -1
            return bm
        else:
            raise ValueError("Invalid dimension or missing faces")


def rgb_to_lab(bgr_colors):
    # 使用OpenCV的正确转换流程
    rgb = cv2.cvtColor(bgr_colors.reshape(1,-1,3), cv2.COLOR_BGR2RGB)
    lab = cv2.cvtColor(rgb, cv2.COLOR_RGB2LAB)
    return lab.reshape(-1,3).astype(np.float32)/255.0


def compute_homology(chain_complex):
    try:
        st = gd.SimplexTree()
        # 确保至少存在1维结构
        if len(chain_complex.dim1) == 0:
            return []
        for i in range(len(chain_complex.dim0)):
            st.insert([i])
        for e in chain_complex.dim1:
            st.insert(e)
        if chain_complex.dim2 is not None:
            for f in chain_complex.dim2:
                st.insert(f)
        st.compute_persistence()
        H1 = [interval for (dim, interval) in st.persistence() if dim == 1]
        return H1 if H1 else [(0.0, 1.0)]  # 返回默认值
    except Exception as e:
        print(f"同调计算异常: {str(e)}")
        return [(0.0, 1.0)]  # 返回默认值避免后续计算错误

class ColorWeight(nn.Module):
    def __init__(self, beta=0.1):
        super().__init__()
        self.beta = nn.Parameter(torch.tensor(beta))

    def lab_distance(self, c1, c2):
        return torch.norm(c1 - c2, dim=-1)


    # 在ColorWeight类中：
    def anisotropic_diffusion(self, W, num_iter=3):
        # 使用PyTorch实现的各向异性扩散
        kernel = torch.tensor([[0.05, 0.2, 0.05],
                               [0.2, -1.0, 0.2],
                               [0.05, 0.2, 0.05]], device=W.device)
        for _ in range(num_iter):
            W = F.conv2d(W.unsqueeze(0), kernel.unsqueeze(0), padding=1)
        return W.squeeze()
    def forward(self, vertices, colors, edge_index):
        src, dst = edge_index
        edge_diff = self.lab_distance(colors[src], colors[dst])
        with torch.no_grad():
            edge_diff = self.anisotropic_diffusion(edge_diff, edge_index)

        return to_dense_adj(
            edge_index,
            edge_attr=torch.exp(-self.beta * edge_diff),
            max_num_nodes=vertices.size(0)
        ).squeeze(0)


class DynamicBatcher:
    def __init__(self, max_mem_usage=0.8):
        self.max_mem = psutil.virtual_memory().total * max_mem_usage

    def auto_batch_size(self, dataset):
        sample_size = asizeof.asizeof(dataset[0]) if len(dataset) > 0 else 0
        if sample_size == 0:
            return 1
        return max(1, int(self.max_mem // sample_size))

class EnhancedColorWeight(nn.Module):
    def __init__(self, in_channels, beta=0.1):
        super().__init__()
        self.beta = nn.Parameter(torch.tensor(beta))
        self.conv = nn.Sequential(
            nn.Conv1d(in_channels, 16, 1),
            nn.ReLU(),
            nn.Conv1d(16, 1, 1)
        )

    def forward(self, vertices, colors, edge_index):
        src, dst = edge_index
        color_diff = torch.abs(colors[src] - colors[dst])
        spatial_diff = torch.norm(vertices[src] - vertices[dst], dim=1)

        features = torch.stack([color_diff, spatial_diff], dim=1)
        weights = self.conv(features.unsqueeze(0)).squeeze()

        return to_dense_adj(
            edge_index,
            edge_attr=torch.sigmoid(weights),
            max_num_nodes=vertices.size(0)
        ).squeeze(0)


class HomologyAnalyzer:
    def __init__(self, persistence_threshold=5.0):
        self.persistence_threshold = persistence_threshold

    def compute_persistent_homology(self, adj_matrix):
        """基于邻接矩阵计算持续同调"""
        st = gd.SimplexTree()

        # 构建邻接矩阵到单纯复形
        np_adj = adj_matrix.cpu().numpy() if isinstance(adj_matrix, torch.Tensor) else adj_matrix
        n = np_adj.shape[0]

        # 添加0-单形
        for i in range(n):
            st.insert([i])

        # 添加1-单形（边）
        for i in range(n):
            for j in range(i + 1, n):
                if np_adj[i, j] > 0.5:
                    st.insert([i, j])

        # 计算持续同调
        st.compute_persistence()
        return st.persistence()

    def extract_generators(self, st):
        """提取重要生成元"""
        generators = []
        for interval in st.persistence():
            dim, (birth, death) = interval
            if dim == 1 and (death - birth) > self.persistence_threshold:
                cycle = st.persistence_intervals_in_dimension(dim)[0]
                generators.append({
                    'dimension': dim,
                    'birth': birth,
                    'death': death,
                    'cycle': cycle
                })
        return generators


class TopologyRepair(nn.Module):
    def __init__(self, in_channels=32, beta=0.5):
        super().__init__()
        self.homology_analyzer = HomologyAnalyzer()
        self.repair_config = {
            'smoothing_iters': 3,
            'dijkstra_threshold': 0.3,
            'persistence_threshold': 5.0
        }
        self.use_attention = True  # 确定是否使用注意力机制（实际未使用该标志）

        # 先初始化注意力层
        self.attention = nn.MultiheadAttention(
            embed_dim=in_channels,  # 输入特征维度
            num_heads=4,  # 注意力头数
            dropout=0.1
        )

        # 后初始化权重参数
        nn.init.xavier_uniform_(self.attention.in_proj_weight)
        nn.init.constant_(self.attention.out_proj.bias, 0.0)

        self.temporal_aggregator = nn.GRU(
            input_size=64,
            hidden_size=64,
            num_layers=2
        )
        self.edge_weight = nn.Sequential(
            nn.Linear(70, 128),
            nn.LayerNorm(128),
            nn.ReLU(),
            nn.Dropout(0.3),
            nn.Linear(128, 64),
            nn.LayerNorm(64),
            nn.ReLU(),
            nn.Linear(64, 1),
            nn.Sigmoid()
        )

        # 初始化权重（原代码保持不变）
        for layer in self.edge_weight:
            if isinstance(layer, nn.Linear):
                nn.init.kaiming_normal_(layer.weight, mode='fan_out', nonlinearity='relu')
                nn.init.constant_(layer.bias, 0.1)

        # 其他参数初始化
        self.cycle_loss_weight = 0.5
        self.edge_ema = None
        self.ema_alpha = 0.9
        self.temporal_loss_weight = 0.3
        self.prev_adj = None
        # 增加图卷积层
        self.conv1 = GCNConv(in_channels, 64)
        self.conv2 = GCNConv(64, 32)

    # 在模型的forward方法中，确保adj_matrix的维度正确
    def forward(self, data):
        x = F.relu(self.conv1(data.x, data.edge_index))
        x = F.dropout(x, p=0.2, training=self.training)
        x = self.conv2(x, data.edge_index)
        batch_size = data.num_graphs if hasattr(data, 'num_graphs') else 1
        # 构造边特征时增加维度检查
        src, dst = data.edge_index
        if src.max() >= data.num_nodes or dst.max() >= data.num_nodes:
            raise ValueError("边索引包含非法节点索引")

        # 计算边权重
        edge_features = self._construct_edge_features(x, data)
        edge_weights = self.edge_weight(edge_features).squeeze()

        # 生成邻接矩阵时明确指定维度
        adj_matrix = to_dense_adj(
            data.edge_index,
            edge_attr=edge_weights,
            max_num_nodes=data.num_nodes
        ).squeeze(0)
        edges = self.update_dynamic_edges(adj_matrix)  # 新增边生成
        # 生成目标邻接矩阵
        adj_target = to_dense_adj(
            data.edge_index,
            max_num_nodes=data.num_nodes
        ).squeeze(0).float()

        # 确保维度匹配
        if adj_matrix.dim() == 1:
            adj_matrix = adj_matrix.unsqueeze(0)
        if adj_matrix.shape != adj_target.shape:
            raise ValueError(f"维度不匹配: {adj_matrix.shape} vs {adj_target.shape}")

        # 计算损失
        recon_loss = F.binary_cross_entropy(adj_matrix, adj_target)

        # 计算环形损失
        cycle_loss = self.calculate_cycle_loss(adj_matrix)

        # 总损失
        total_loss = recon_loss + 0.1 * cycle_loss

        return {
            'adj_matrix': adj_matrix,
            'edges': edges,
            'vertices': data.pos,  # 添加顶点位置信息
            'loss': total_loss
        }
    def _forward_batch(self, batch):
        """处理批次数据"""
        x = batch.x
        edge_index = batch.edge_index
        colors = batch.colors
        batch_vector = batch.batch  # 获取图的批次索引

        # 计算空间和颜色差异（支持批次）
        src, dst = edge_index
        spatial_dist = torch.norm(x[src] - x[dst], dim=1, keepdim=True)
        color_diff = colors[src] - colors[dst]
        # 构建边特征（自动处理批次维度）
        edge_features = torch.cat([
            x[src], x[dst],  # 32 + 32 = 64
            spatial_dist,  # 1
            color_diff,  # 3 (假设color_diff是3维的Lab颜色差)
            color_diff.norm(dim=1, keepdim=True),  # 1
            torch.ones_like(spatial_dist)  # 1
        ], dim=1)  # 总维度 = 64 + 1 + 3 + 1 + 1 = 70

        # 计算边权重
        edge_weights = torch.sigmoid(self.edge_weight(edge_features)).squeeze()

        # 使用PyG内置方法创建批次邻接矩阵
        adj_matrix = to_dense_adj(
            edge_index,
            edge_attr=edge_weights,
            batch=batch_vector
        )

        # 计算损失（自动广播到批次）
        recon_loss = F.binary_cross_entropy(
            adj_matrix,
            torch.ones_like(adj_matrix)
        )
        cycle_loss = self.calculate_cycle_loss(adj_matrix)

        return {
            'edges': self.update_dynamic_edges(adj_matrix),
            'weights': adj_matrix,
            'loss': recon_loss + 0.1 * cycle_loss
        }

    def _construct_edge_features(self, x, data):
        src, dst = data.edge_index
        spatial_dist = torch.norm(x[src] - x[dst], dim=1, keepdim=True)
        color_diff = data.colors[src] - data.colors[dst]

        edge_features = torch.cat([
            x[src], x[dst],  # 32+32=64
            spatial_dist,  # 1
            color_diff,  # 3
            torch.norm(color_diff, dim=1, keepdim=True),  # 1
            torch.ones_like(spatial_dist)  # 1
        ], dim=1)  # 总维度70

        return edge_features
    def _forward_single(self, data):
        return self.process_single_graph(data.x, data.edge_index, data.colors)

    def process_batch(self, batch):
        outputs = []
        for graph in batch.to_data_list():
            outputs.append(self.process_single_graph(graph))  # 更正方法名
        return self.merge_outputs(outputs)

    def process_single_graph(self, data):
        """处理单个图结构"""
        try:
            x = data.x
            edge_index = data.edge_index
            colors = data.colors

            # 原process_single_graph的处理逻辑
            if x.size(0) == 0 or edge_index.size(1) == 0:
                raise ValueError("空图输入")

            # 边索引验证
            if edge_index.max() >= x.size(0):
                raise ValueError("边索引超出节点范围")
            # 计算空间和颜色差异
            src, dst = edge_index
            spatial_dist = torch.norm(x[src] - x[dst], dim=1, keepdim=True)
            color_diff = colors[src] - colors[dst]  # 直接使用颜色差值(3维)

            # 在 process_single_graph 方法中检查特征拼接
            edge_features = torch.cat([
                x[src], x[dst],  # 32 + 32 = 64
                spatial_dist,  # 1
                color_diff,  # 3 (LAB颜色差异)
                color_diff.norm(dim=1, keepdim=True),  # 1
                torch.ones_like(spatial_dist)  # 1
            ], dim=1)  # 总维度 = 64 + 1 + 3 + 1 + 1 = 70
            if edge_features.shape[1] != 70:
                raise ValueError(f"无效的特征维度: {edge_features.shape}")

            edge_weights = self.edge_weight(edge_features).squeeze()
            adj_matrix = to_dense_adj(
                edge_index,
                edge_attr=edge_weights,
                max_num_nodes=x.size(0)
            ).squeeze(0)
            # 对称化处理
            adj_matrix = (adj_matrix + adj_matrix.T) / 2

            adj_target = torch.zeros_like(adj_matrix)
            adj_target[edge_index[0], edge_index[1]] = 1.0
            adj_target = (adj_target + adj_target.T) / 2  # 确保对称

            # 重建损失计算
            recon_loss = F.binary_cross_entropy(
                adj_matrix,
                adj_target  # 使用Delaunay生成的边作为目标
            )

            # 计算环形损失
            cycle_loss = self.calculate_cycle_loss(adj_matrix)

            # 更新动态边
            updated_edges = self.update_dynamic_edges(adj_matrix)

            loss = recon_loss + 0.1 * cycle_loss
            loss.requires_grad_(True)  # 确保梯度可导
            return {
                'vertices': data.pos,  # 节点坐标 [num_nodes, 2]
                'edges': updated_edges.t().contiguous(),  # 边索引 [2, num_edges]
                'weights': adj_matrix,
                'loss': loss
            }

        except Exception as e:
            print(f"子图处理失败: {str(e)}")
            # 返回空结构
            return {
                'edges': torch.zeros(2, 0, dtype=torch.long),
                'weights': torch.zeros(0, 0),
                'loss': torch.tensor(0.0)
            }

    def update_dynamic_edges(self, adj_matrix, dynamic_threshold=True):
        """动态阈值调整的边更新"""
        # 自动阈值计算
        if dynamic_threshold:
            valid_weights = adj_matrix[adj_matrix > 0]
            threshold = torch.mean(valid_weights) * 0.6 if len(valid_weights) > 0 else 0.3
        else:
            threshold = 0.3

        # 生成对称邻接矩阵
        sym_adj = (adj_matrix + adj_matrix.T) / 2
        edges = (sym_adj > threshold).nonzero(as_tuple=False).t()

        # 确保最小连接
        if edges.size(1) == 0 and adj_matrix.size(0) >= 2:
            edges = torch.combinations(
                torch.arange(adj_matrix.size(0)),
                r=2
            ).t()

        return edges

    def smooth_weights(self, W, edge_index, num_iter=3):
        """各向异性扩散滤波"""
        try:
            import cv2.ximgproc
            device = W.device
            W_np = W.cpu().numpy()
            W_norm = cv2.normalize(W_np, None, 0, 255, cv2.NORM_MINMAX).astype(np.uint8)

            for _ in range(num_iter):
                W_norm = cv2.ximgproc.anisotropicDiffusion(
                    W_norm, alpha=0.1, K=15, niters=1
                )

            W_smoothed = cv2.normalize(W_norm, None, 0, 1, cv2.NORM_MINMAX)
            return torch.from_numpy(W_smoothed).float().to(device)
        except Exception as e:
            print(f"使用OpenCV滤波失败，启用备用方案: {str(e)}")
            for _ in range(num_iter):
                W = (W + W.mean(dim=1, keepdim=True)) / 2
            return W

    def update_dynamic_edges(self, W, threshold=0.3):  # 降低阈值
        edges = (W > threshold).nonzero(as_tuple=False).t()
        if edges.size(1) == 0 and W.size(0) >= 2:
            edges = torch.tensor([[0, 1]], dtype=torch.long).t()
        return edges

    def calculate_cycle_loss(self, adj_matrix):
        # 构造单纯复形
        st = gd.SimplexTree()
        indices = torch.where(adj_matrix > 0.5)
        # 添加边到复形...
        st.compute_persistence()

        # 计算持久环损失
        persistence = st.persistence()
        h1_life = [d - b for dim, (b, d) in persistence if dim == 1]
        # 此处，确保返回张量
        h1_tensor = torch.tensor(h1_life) if h1_life else torch.tensor([0.0], device=adj_matrix.device)
        return -torch.log(torch.mean(h1_tensor) + 1e-7)  # 添加平滑项防止log(0)
    def repair_contour(self, adj_matrix, pos, colors):
        """实现论文算法1的完整流程"""
        # Step 1: 提取同调生成元
        st = self.homology_analyzer.compute_persistent_homology(adj_matrix)
        generators = self.homology_analyzer.extract_generators(st)

        # Step 2: 支撑集提取
        edge_masks = self.extract_support(adj_matrix, generators)

        # Step 3: 局部权重平滑
        smoothed_weights = self.anisotropic_smoothing(
            adj_matrix,
            iterations=self.repair_config['smoothing_iters']
        )

        # Step 4: 连接断裂端点
        repaired_adj = self.connect_endpoints(
            smoothed_weights,
            pos,
            threshold=self.repair_config['dijkstra_threshold']
        )

        # Step 5: 拓扑约束优化
        final_adj = self.topology_aware_refinement(
            repaired_adj,
            generators,
            color_weights=colors
        )

        return final_adj

    def anisotropic_smoothing(self, W, iterations=3):
        """各向异性扩散的改进实现"""
        for _ in range(iterations):
            # 使用论文中的权重更新公式
            W = 0.5 * W + 0.5 * F.avg_pool2d(W.unsqueeze(0), 3, padding=1, stride=1).squeeze()
            W = torch.clamp(W, 0, 1)
        return W

    def connect_endpoints(self, adj, pos, threshold=0.3):
        """基于Dijkstra的断裂连接"""
        device = adj.device
        n = adj.shape[0]
        pos_np = pos.cpu().numpy()

        # 构建图结构
        G = nx.Graph()
        for i in range(n):
            for j in range(i + 1, n):
                if adj[i, j] > threshold:
                    G.add_edge(i, j, weight=1 / adj[i, j])

        # 寻找断裂组件
        components = list(nx.connected_components(G))

        # 连接最近邻组件
        for i in range(len(components)):
            for j in range(i + 1, len(components)):
                # 寻找组件间最近点对
                min_dist = float('inf')
                pair = None
                for u in components[i]:
                    for v in components[j]:
                        dist = np.linalg.norm(pos_np[u] - pos_np[v])
                        if dist < min_dist:
                            min_dist = dist
                            pair = (u, v)

                if pair is not None:
                    # 添加新边
                    adj[pair[0], pair[1]] = 1.0
                    adj[pair[1], pair[0]] = 1.0

        return adj

    def topology_aware_refinement(self, adj, generators, color_weights):
        """拓扑约束优化"""
        # 实现论文中的能量最小化公式
        lap = torch.diag(adj.sum(1)) - adj
        color_sim = torch.exp(-torch.cdist(color_weights, color_weights))

        # 构建优化目标
        objective = 0.7 * adj + 0.2 * lap + 0.1 * color_sim

        # 应用生成元约束
        for gen in generators:
            cycle = gen['cycle']
            n = len(cycle)
            for i in range(n):
                u = cycle[i]
                v = cycle[(i + 1) % n]
                objective[u, v] += 0.5  # 增强循环约束

        return torch.sigmoid(objective)

    def calculate_total_loss(self, adj_matrix):
        # 基于Betti数的拓扑损失
        st = gd.SimplexTree()
        indices = torch.where(adj_matrix > 0.5)
        for i, j in zip(indices[0], indices[1]):
            if i < j:  # 避免重复添加
                st.insert([i.item(), j.item()])
        st.compute_persistence()

        # 计算H1生命周期
        h1_life = [d - b for (dim, (b, d)) in st.persistence() if dim == 1]
        h1_loss = -torch.log(torch.tensor(sum(h1_life) + 1e-7))  # 最大化生命周期

        # 节点度分布正则化
        degrees = adj_matrix.sum(dim=1)
        degree_loss = torch.var(degrees)  # 鼓励均衡连接

        # 稀疏性约束
        density = torch.mean(adj_matrix)
        sparsity_loss = torch.abs(density - 0.15)  # 目标连接密度15%

        return 0.5 * h1_loss + 0.3 * degree_loss + 0.2 * sparsity_loss

class WeightSystemValidator:
    def __init__(self, lambda_reg=0.1):
        self.lambda_reg = lambda_reg

    def compute_discrete_curvature(self, W, edge_index):
        """计算离散曲率张量"""
        curvature = {}
        for u, v in edge_index.t().tolist():
            # 寻找包含u-v的三角形
            neighbors = set(torch.where(edge_index[0] == u)[0]) & set(torch.where(edge_index[1] == v)[0])
            for w in neighbors:
                # 检查三角形闭合性
                if (u, w) in edge_index and (v, w) in edge_index:
                    # 计算三角形曲率
                    curvature[(u, v, w)] = W[u, v] * W[v, w] * W[w, u] - 1
        return curvature

    def validate_flatness(self, W, edge_index):
        """验证权重系统的平坦性条件"""
        curvature = self.compute_discrete_curvature(W, edge_index)
        total_violation = sum(abs(c) for c in curvature.values())
        return total_violation / (len(curvature) + 1e-7)

    def optimize_flatness(self, W, edge_index, lr=0.01, steps=100):
        """优化权重平坦性"""
        W = W.clone().requires_grad_(True)
        optimizer = torch.optim.Adam([W], lr=lr)

        for _ in range(steps):
            optimizer.zero_grad()

            # 计算曲率损失
            curvature = self.compute_discrete_curvature(W, edge_index)
            loss = sum(c ** 2 for c in curvature.values())

            # 正则化项
            reg_loss = self.lambda_reg * torch.norm(W - 0.5)

            total_loss = loss + reg_loss
            total_loss.backward()
            optimizer.step()

            # 保持权重在合理范围
            with torch.no_grad():
                W.data = torch.clamp(W, 0.1, 0.9)

        return W.detach()

class FibrationValidator:
    def __init__(self, epsilon=1e-3):
        self.epsilon = epsilon

    def check_fibration_continuity(self, pi, X, Y, Z):
        """验证纤维化映射π的连续性"""
        # 在Y和Z的交集附近采样
        samples = self._sample_near_intersection(Y, Z)

        # 检查局部截面的存在性
        for x in samples:
            U_x = self.find_consistent_neighborhood(x, pi)
            if not self.verify_local_section(pi, U_x):
                return False

        return True

    def _sample_near_intersection(self, Y, Z, n_samples=100):
        """在Y∩Z附近生成采样点"""
        # 使用Halton序列进行准蒙特卡洛采样
        sampler = qmc.Halton(d=3, scramble=True)
        samples = sampler.random(n_samples)
        return Y.bounds[0] + samples * (Y.bounds[1] - Y.bounds[0])

    def verify_local_section(self, pi, neighborhood):
        try:
            # 使用逆映射构造截面
            inv_map = self.construct_inverse(pi, neighborhood)
            return self.check_section_consistency(pi, inv_map)
        except NonInvertibleError:  # 现在引用的是模块级别的异常类
            return False

    def construct_inverse(self, pi, U):
        """基于深度偏序构造逆映射"""
        # 实现论文中的局部截面构造方法
        depth_sorted = sorted(U, key=lambda x: pi(x)[1])  # 假设π返回(坐标,深度)
        return {p: depth_sorted[0] for p in U}

# 在EnhancedSpatioTemporalLoss类之前添加SpatioTemporalLoss基类
class SpatioTemporalLoss(nn.Module):
    def __init__(self):
        super().__init__()
        self.temporal_loss = nn.MSELoss()

    def forward(self, current_adj, prev_adj):
        spatial_loss = 1 - F.cosine_similarity(current_adj, prev_adj).mean()
        return spatial_loss + 0.3 * self.temporal_loss(current_adj, prev_adj)


class EnhancedSpatioTemporalLoss(SpatioTemporalLoss):
    def __init__(self):
        super().__init__()
        # 增加运动预测模块
        self.lstm = nn.LSTM(input_size=2, hidden_size=32, num_layers=2)
        self.attention = nn.MultiheadAttention(embed_dim=32, num_heads=4)

    def forward(self, current_adj, prev_adj, motion_seq):
        # 时序建模
        motion_seq = motion_seq.permute(2, 0, 1)  # [seq_len, batch, features]
        lstm_out, _ = self.lstm(motion_seq)
        attn_out, _ = self.attention(lstm_out, lstm_out, lstm_out)
        pred_flow = attn_out[-1]  # 获取最新预测

        # 多模态损失计算
        flow_loss = F.mse_loss(pred_flow, motion_seq[-1])
        spatial_loss = 1 - F.cosine_similarity(current_adj, prev_adj).mean()

        return 0.6 * flow_loss + 0.4 * spatial_loss


# 辅助函数
class TopoGAN(nn.Module):
    def __init__(self, latent_dim=256):
        super().__init__()
        self.generator = nn.Sequential(
            nn.Linear(latent_dim, 512),
            nn.BatchNorm1d(512),
            nn.LeakyReLU(0.2),
            nn.Linear(512, 1024),
            nn.Tanh()  # 输出归一化坐标
        )

        self.top_loss = TopologicalLoss()

    def forward(self, z):
        # 生成点云 [batch_size, num_points, 3]
        points = self.generator(z).view(-1, 1024, 3)

        # 判别器损失
        real_pred = self.discriminator(points)
        adv_loss = F.binary_cross_entropy_with_logits(real_pred, torch.ones_like(real_pred))

        # 拓扑损失
        topo_loss = self.top_loss(points)

        return adv_loss + 0.3 * topo_loss


class TopologicalLoss(nn.Module):
    def forward(self, points):
        losses = []
        for batch in points:
            # 构建Alpha复形
            alpha = gd.AlphaComplex(points=batch.cpu().numpy())
            st = alpha.create_simplex_tree()
            st.compute_persistence()

            # 获取Betti数并安全访问
            betti = st.betti_numbers()
            beta1 = betti[1] if len(betti) > 1 else 0  # 处理维度缺失情况

            # 计算H1生命周期（显式检查维度）
            h1 = [death - birth for (dim, (birth, death)) in st.persistence() if dim == 1]
            h1_mean = np.mean(h1) if h1 else 0.0

            # 组合损失项（同时考虑Betti数和生命周期）
            losses.append(-h1_mean + torch.abs(torch.tensor(beta1) - 1).item())

        return torch.mean(torch.tensor(losses, device=points.device))

                })

    def _track_features(self, gray_frame):
        """改进的运动一致性验证"""
        # 初始化时空窗口
        if not hasattr(self, 'prev_gray_deque'):
            self.prev_gray_deque = deque(maxlen=3)
            self.prev_kp_deque = deque(maxlen=3)

        # 多尺度运动验证
        all_flows = []
        for level in range(3):
            scaled_gray = cv2.resize(gray_frame, None, fx=1 / (2 ** level), fy=1 / (2 ** level))
            flow = cv2.calcOpticalFlowFarneback(
                self.prev_gray_deque[-1], scaled_gray,
                None, 0.5, 3, 15, 3, 5, 1.2, 0
            )
            all_flows.append(cv2.resize(flow, gray_frame.shape[::-1]))

        # 运动场融合
        fused_flow = np.mean(all_flows, axis=0)
        return self._validate_motion_consistency(fused_flow)

    def _analyze_topology(self, frame, keypoints):
        """增强的拓扑风险分析"""
        # 构建多粒度链复形
        complex_levels = []
        for scale in [1.0, 0.5, 0.25]:
            scaled_kps = keypoints * scale
            edges = self._delaunay_triangulation(scaled_kps)
            complex_levels.append(ChainComplex(scaled_kps, edges))

        # 多尺度持续同调分析
        risk_scores = []
        for complx in complex_levels:
            H1 = compute_homology(complx)
            lifetimes = [d - b for (b, d) in H1]
            risk_scores.append(len([t for t in lifetimes if t > 5]))

        # 动态风险评估
        if np.mean(risk_scores) > 2.5:
            self._trigger_emergency_alert(frame)
            self._update_risk_map(keypoints)

    def _mark_danger_zone(self, frame, keypoints, cycles):
        # 标记危险区域
        mask = np.zeros_like(frame)
        for cycle in cycles:
            pts = keypoints[cycle].astype(int)
            hull = cv2.convexHull(pts)
            cv2.fillPoly(mask, [hull], (0, 0, 255))

        # 显示警告
        frame = cv2.addWeighted(frame, 0.7, mask, 0.3, 0)
        cv2.putText(frame, "DANGER ZONE!", (50, 80),
                    cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 0, 255), 3)
        return frame

        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

    def project_pointcloud(points, calib):
        """将点云投影到图像平面"""
        # 转换到齐次坐标
        points_homo = np.hstack([points[:, :3], np.ones((points.shape[0], 1))])

        R_rect = np.eye(4)

        # 坐标变换
        points_cam = R_rect @ Tr_velo_to_cam @ points_homo.T
        points_cam = points_cam[:3, :]

        # 投影到图像平面
        P2 = calib['P2'].reshape(3, 4)
        points_img = P2 @ np.vstack([points_cam, np.ones((1, points_cam.shape[1]))])
        points_img = (points_img[:2] / points_img[2]).T

        # 过滤无效点
        valid = (points_img[:, 0] >= 0) & (points_img[:, 0] < 1242) & \
                (points_img[:, 1] >= 0) & (points_img[:, 1] < 375)

        return points_img[valid]  # 返回投影后的有效点坐标

    def _build_pyramid_features(self, frame):
        """构建多尺度时空金字塔特征

        Args:
            frame (np.ndarray): 当前帧图像(BGR格式)

        Returns:
            list: 各金字塔层的时序特征向量
        """
        # 初始化空间金字塔
        pyramids = [frame.copy()]

        # 生成空间金字塔（3层：原图 + 2次下采样）
        for _ in range(2):
            # 下采样前检查最小尺寸
            h, w = pyramids[-1].shape[:2]
            if h <= 20 or w <= 20:
                break
            downsampled = cv2.pyrDown(pyramids[-1])
            pyramids.append(downsampled)

        temporal_features = []

        # 各层级联处理
        for level, img in enumerate(pyramids):
            # 确保特征金字塔初始化
            if f'level{level}' not in self.feature_pyramids:
                self.feature_pyramids[f'level{level}'] = deque(maxlen=5)  # 保留最近5帧

            # 计算光流
            if len(self.feature_pyramids[f'level{level}']) > 0:
                # 获取前一帧同层特征
                prev_frame = self.feature_pyramids[f'level{level}'][-1][..., 0]  # 提取图像部分
                flow = cv2.calcOpticalFlowFarneback(
                    prev_frame, cv2.cvtColor(img, cv2.COLOR_BGR2GRAY),
                    None, 0.5, 3, 15, 3, 5, 1.1, 0
                )
                # 转换为RGB流场可视化
                flow_rgb = self._flow_to_rgb(flow)
            else:
                # 首帧初始化
                flow_rgb = np.zeros_like(img)

            # 提取HOG特征
            hog_feat = self._calc_hog_features(img)

            # 特征拼接 (原图 + 光流 + HOG)
            combined = np.concatenate([
                img,  # 原图特征 (H,W,3)
                flow_rgb,  # 光流场 (H,W,3)
                hog_feat[..., np.newaxis].repeat(3, axis=-1)  # HOG扩展为3通道 (H,W,3)
            ], axis=-1)  # 最终维度 (H,W,9)

            # 更新特征金字塔
            self.feature_pyramids[f'level{level}'].append(combined)

            # 计算本层时序特征（均值池化）
            layer_feat = np.mean(combined, axis=(0, 1))  # 形状(9,)
            temporal_features.append(layer_feat)

        return temporal_features

    def _flow_to_rgb(self, flow):
        """将光流场转换为RGB可视化"""
        hsv = np.zeros((flow.shape[0], flow.shape[1], 3), dtype=np.uint8)
        mag, ang = cv2.cartToPolar(flow[..., 0], flow[..., 1])
        hsv[..., 0] = ang * 180 / np.pi / 2
        hsv[..., 1] = 255
        hsv[..., 2] = cv2.normalize(mag, None, 0, 255, cv2.NORM_MINMAX)
        return cv2.cvtColor(hsv, cv2.COLOR_HSV2BGR)

    def _calc_hog_features(self, img, cell_size=(8, 8)):
        """计算HOG特征图"""
        gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

        # 计算梯度
        gx = cv2.Sobel(gray, cv2.CV_32F, 1, 0)
        gy = cv2.Sobel(gray, cv2.CV_32F, 0, 1)

        # 计算幅值和方向
        mag, ang = cv2.cartToPolar(gx, gy)
        ang = np.rad2deg(ang) % 180

        # 构建HOG矩阵
        hog_matrix = np.zeros((gray.shape[0] // cell_size[0],
                               gray.shape[1] // cell_size[1],
                               9), dtype=np.float32)

        # 分cell统计
        for i in range(0, gray.shape[0], cell_size[0]):
            for j in range(0, gray.shape[1], cell_size[1]):
                cell_mag = mag[i:i + cell_size[0], j:j + cell_size[1]]
                cell_ang = ang[i:i + cell_size[0], j:j + cell_size[1]]

                # 构建直方图
                hist, _ = np.histogram(cell_ang, bins=9, range=(0, 180), weights=cell_mag)
                hog_matrix[i // cell_size[0], j // cell_size[1]] = hist

        # 归一化处理
        eps = 1e-7
        hog_matrix = hog_matrix / (np.linalg.norm(hog_matrix, axis=2, keepdims=True) + eps)

        return hog_matrix

    def process_frame(self, frame):
        graph = self._build_graph(frame)
        if graph is None or graph.edge_index.size(1) == 0:
            print("[警告] 空图结构，跳过处理")
            return frame

        with torch.no_grad():
            output = self.model(
                x=graph.x.to(self.device),
                edge_index=graph.edge_index.to(self.device),
                colors=graph.colors.to(self.device)
            )
        # 添加位置信息到输出
        print(f"[调试] 预测边数量: {output['edges'].size(1)}")
        print(
            f"[调试] 权重统计 - 最大: {output['weights'].max().item():.2f} 平均: {output['weights'].mean().item():.2f}")
        output['pos'] = graph.pos.to(self.device)
        return self._post_process(output, frame, graph)

    def _build_graph(self, frame):
        # 与VideoProcessor中的_build_graph相同实现
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)

        # 特征检测和描述符计算
        kp = self.detector.detect(gray)
        kp = sorted(kp, key=lambda x: -x.response)[:100]  # 取前100个关键点
        kp, descriptors = self.detector.detectAndCompute(gray, None)
        # 处理无描述符情况
        if descriptors is None:
            descriptors = np.zeros((0, 32), dtype=np.float32)
        else:
            # 确保描述符维度正确
            descriptors = descriptors.astype(np.float32)[:, :32]  # 截断或填充至32维
        # 选择响应最高的关键点
        if len(kp) > 100:
            indices = np.argsort([-k.response for k in kp])[:100]
            kp = [kp[i] for i in indices]
            descriptors = descriptors[indices]
        if frame is None or frame.size == 0:
            raise ValueError("输入帧为空")

        h, w = frame.shape[:2]
        if h != self.frame_size[1] or w != self.frame_size[0]:
            frame = cv2.resize(frame, self.frame_size)  # 确保尺寸一致
        # 转换为坐标数组
        pos = np.array([(p.pt[0], p.pt[1]) for p in kp], dtype=np.float32)

        # Delaunay三角剖分构建边
        edges = delaunay_triangulation(pos, self.frame_size)

        # 提取颜色信息
        colors = frame[[int(p[1]) for p in pos], [int(p[0]) for p in pos]]
        colors = frame[[int(p[1]) for p in pos], [int(p[0]) for p in pos]]  # 保持BGR格式
        # 转换到Lab颜色空间
        colors = cv2.cvtColor(colors, cv2.COLOR_BGR2LAB)  # 关键点
        colors = colors.astype(np.float32) / 255.0  # 归一化
        colors = rgb_to_lab(colors)
        if edges.shape[1] == 0:  # 无有效边
            if len(pos) >= 2:
                # 创建默认边（连接前两个点）
                edges = torch.tensor([[0, 1]], dtype=torch.long).t().contiguous()
            else:
                # 没有足够点，返回空边
                edges = torch.zeros((2, 0), dtype=torch.long)
        edges = delaunay_triangulation(pos, (w, h))
        print(f"生成边数量: {edges.shape[1]}")
        return Data(
            x=torch.tensor(descriptors, dtype=torch.float32),
            pos=torch.tensor(pos, dtype=torch.float32),
            edge_index=edges,
            colors=torch.tensor(colors, dtype=torch.float32) / 255.0
        )

    def _post_process(self, output, frame, graph):
        overlay = frame.copy()
        if 'edges' in output:
            pos = output['pos'].cpu().numpy()
            for u, v in graph.edge_index.t().cpu().numpy():
                pt1 = tuple(map(int, pos[u]))
                pt2 = tuple(map(int, pos[v]))
                cv2.line(overlay, pt1, pt2, (255, 0, 0), 1)  # 蓝色显示原始边

            adj_matrix = output['weights'].cpu().numpy()  # 获取邻接矩阵
            edges = output['edges'].t().cpu().numpy()

            # 计算运动方向一致性
            motion_vectors = pos[edges[:, 1]] - pos[edges[:, 0]]
            motion_magnitude = np.linalg.norm(motion_vectors, axis=1)
            motion_direction = motion_vectors / (motion_magnitude[:, None] + 1e-5)

            # 保留运动方向一致的边（夹角小于30度）
            for i, (u, v) in enumerate(edges):
                # 正确获取权重值
                weight = adj_matrix[u, v]

                if weight < 0.3:
                    continue
                if i > 0:
                    angle_threshold = np.pi / 4  # 30度
                    dot_product = np.clip(..., -1.0, 1.0)
                    if np.arccos(dot_product) > angle_threshold:
                        continue
                pt1 = tuple(map(int, pos[u]))
                pt2 = tuple(map(int, pos[v]))
                cv2.line(overlay, pt1, pt2, (0, 255, 0), 2)
            # 绘制拓扑特征热力图
            plt.figure(figsize=(12, 6))
            plt.imshow(output['weights'].cpu().numpy(), cmap='hot')
            plt.colorbar()
            plt.savefig('topo_heatmap.png')
            plt.close()

            return overlay
        return cv2.addWeighted(frame, 0.7, overlay, 0.3, 0)

    def _fuse_semantic_segmentation(self, frame):
        """轻量级语义分割融合"""
        # 初始化分割模型
        if not hasattr(self, 'seg_net'):
            self.seg_net = cv2.dnn.readNet(
                'fcn_resnet50.onnx'  # 预训练的轻量分割模型
            )

        # 执行实时推理
        blob = cv2.dnn.blobFromImage(frame, 1 / 255., (512, 512), swapRB=True)
        self.seg_net.setInput(blob)
        output = self.seg_net.forward()

        # 提取道路区域
        road_mask = np.argmax(output, axis=1)[0] == 0  # 假设0是道路类别
        return cv2.bitwise_and(frame, frame, mask=road_mask.astype(np.uint8))

    def _compute_descriptors(self, gray, kp):
        """ 计算关键点描述符 """
        # 转换关键点格式为cv2.KeyPoint
        keypoints = [cv2.KeyPoint(x=p[0], y=p[1], _size=10) for p in kp]

        # 使用ORB计算描述符
        _, descriptors = self.detector.compute(gray, keypoints)
        if descriptors is None:
            return np.zeros((len(kp), 32), dtype=np.float32)  # 返回默认描述符

        return descriptors.astype(np.float32)

    def start_processing(self, input_source, output_path=None):
        cap = cv2.VideoCapture(input_source)
        # 动态获取视频参数
        frame_width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
        frame_height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
        fps = cap.get(cv2.CAP_PROP_FPS)
        self.frame_size = (frame_width, frame_height)

        # 初始化视频写入器
        if output_path:
            fourcc = cv2.VideoWriter_fourcc(*'mp4v')
            out = cv2.VideoWriter(output_path, fourcc, fps, (frame_width, frame_height))

        # 处理循环
        while True:
            ret, frame = cap.read()
            if not ret:
                break

            processed = self.process_frame(frame)

            # 写入输出视频
            if output_path:
                out.write(processed)

            # 显示实时画面
            cv2.imshow('Topology Repair', processed)
            if cv2.waitKey(1) & 0xFF == ord('q'):
                break

        # 释放资源
        cap.release()
        if output_path:
            out.release()
        cv2.destroyAllWindows()

    def _verify_sample_files(self, check_num=100):
        """随机抽查文件完整性"""
        import random
        sample_ids = random.sample(self.frame_list, min(check_num, len(self.frame_list)))

        for fid in sample_ids:
            try:
                data = self.__getitem__(self.frame_list.index(fid))
                # 验证关键属性
                assert data.pos.shape[1] == 2, "坐标维度错误"
                assert data.edge_index.shape[0] == 2, "边索引格式错误"
                assert data.x.shape[1] == 32, "特征维度错误"
            except Exception as e:
                print(f"文件校验失败 {fid}: {str(e)}")
                self.frame_list.remove(fid)  # 自动移除非法帧

    def _validate_dataset(self):
        """移除无效帧"""

        for idx in range(len(self.frame_list)):

        calib_path = os.path.join(self.root_dir, seq_id, 'calib', 'calib.txt')
        calib = {}
        with open(calib_path) as f:
            for line in f:
                if ':' in line:
                    key, value = line.split(':', 1)
                    values = [float(v) for v in value.strip().split()]
                    if key.startswith('R'):
                        calib[key.strip()] = np.array(values).reshape(3, 3)
                    else:
                        calib[key.strip()] = np.array(values)
        return calib

    def _detect_keypoints(self, image):
        # 显式指定特征点数量参数
        detector = cv2.ORB_create(nfeatures=100, scoreType=cv2.ORB_FAST_SCORE)
        kp, descriptors = detector.detectAndCompute(image, None)

        # 关键点数量控制
        if len(kp) > 100:
            kp = sorted(kp, key=lambda x: -x.response)[:100]
            if descriptors is not None:
                descriptors = descriptors[:100]

        # 处理无描述符情况
        if descriptors is None:
            return [], np.zeros((0, 32), dtype=np.float32)

        # 统一描述符维度为64
        descriptors = descriptors[:, :64] if descriptors.shape[1] > 64 else np.pad(descriptors, (
        (0, 0), (0, 64 - descriptors.shape[1])))
        return kp, descriptors

    def _build_graph(self, image):
        """构建图结构（与VideoProcessor中的方法保持一致）"""

        h, w = image.shape[:2]

        # 显式指定特征点数量参数
        detector = cv2.ORB_create(nfeatures=100, scoreType=cv2.ORB_FAST_SCORE)
        kp, descriptors = detector.detectAndCompute(image, None)

        max_keypoints = 200
        if len(kp) > max_keypoints:
            indices = np.random.choice(len(kp), max_keypoints, replace=False)
            kp = [kp[i] for i in indices]
            descriptors = descriptors[indices] if descriptors is not None else None

        # 处理无关键点情况
        if descriptors is None:
            descriptors = np.zeros((0, 32), dtype=np.float32)
            kp = []

        descriptors = descriptors[:, :32] if descriptors.shape[1] > 32 else np.pad(
            descriptors, ((0, 0), (0, 32 - descriptors.shape[1])))

        # 坐标转换
        pos = np.array([(p.pt[0], p.pt[1]) for p in kp], dtype=np.float32) if kp else np.zeros((0, 2))
        # Delaunay三角剖分
        edges = delaunay_triangulation(pos, (w, h)) if len(pos) >= 2 else torch.zeros((2, 0), dtype=torch.long)

        # 新增：过滤非法边索引
        if edges.size(1) > 0:
            max_node_idx = pos.shape[0] - 1  # 最大有效节点索引
            valid_mask = (edges[0] <= max_node_idx) & (edges[1] <= max_node_idx)
            edges = edges[:, valid_mask]
        # 颜色特征提取
        colors = np.zeros((len(pos), 3), dtype=np.float32)
        if len(pos) > 0:
            # 提取关键点颜色（BGR格式）
            h, w = image.shape[:2]
            if len(pos) > 0:
                y_coords = np.clip(pos[:, 1].astype(int), 0, h - 1)
                x_coords = np.clip(pos[:, 0].astype(int), 0, w - 1)
                colors = image[y_coords, x_coords]
        if len(kp) < 10:
            print("补充网格关键点")
            h, w = image.shape[:2]
            grid_x = np.linspace(0, w - 1, 10)
            grid_y = np.linspace(0, h - 1, 5)
            grid_pts = np.stack(np.meshgrid(grid_x, grid_y), axis=-1).reshape(-1, 2)
            kp = [cv2.KeyPoint(x=pt[0], y=pt[1], _size=10) for pt in grid_pts]
        if edges.shape[1] > 0:
            max_idx = pos.shape[0] - 1
            edges = edges[:, (edges[0] <= max_idx) & (edges[1] <= max_idx)]
        return Data(
            x=torch.tensor(descriptors, dtype=torch.float32),
            pos=torch.tensor(pos, dtype=torch.float32),
            edge_index=edges,
            colors=torch.tensor(colors, dtype=torch.float32) / 255.0
        )

            if data['image'].shape[:2] != (375, 1242):
                data['image'] = cv2.resize(data['image'], (1242, 375))

            # 构建图结构
            graph = self._build_graph(data['image'])

            # 图结构验证
            if graph.num_nodes < 5 or graph.num_edges < 10:
                raise ValueError("图结构不完整")

            return graph
        except Exception as e:
            print(f"跳过无效帧 {frame_id}: {str(e)}")
            return self._get_fallback_data()

    def _load_frame_data(self, frame_id):
        """封装数据加载与图构建逻辑"""
        data_dir = os.path.join(self.root_dir, self.data_type)
        data = load_kitti_data(frame_id=int(frame_id), data_dir=data_dir)

        # 输入数据验证
        if data['image'].size == 0 or data['pointcloud'].size == 0:
            raise ValueError("空图像或点云数据")

        # 构建图结构
        graph = self._build_graph(data['image'])

        # 图结构验证
        if graph.num_nodes > 500 or graph.num_edges > 10000:
            raise ValueError("图结构过大")
        if graph.edge_index.max() >= graph.num_nodes:
            raise ValueError("边索引越界")

        return data, graph

    def _validate_graph(self, graph):
        """验证图结构基本有效性"""
        return graph.num_nodes >= 3 and graph.num_edges >= 2

    def _visualize_graph(self, graph, frame_id):
        """可视化调试首帧图结构"""
        pos = graph.pos.numpy()
        edges = graph.edge_index.numpy().T
        plt.figure(figsize=(10, 6))
        plt.scatter(pos[:, 0], pos[:, 1], c='r', s=10)
        for u, v in edges:
            plt.plot([pos[u, 0], pos[v, 0]], [pos[u, 1], pos[v, 1]], 'b-', linewidth=0.5)
        plt.title(f'Frame {frame_id} Graph Structure')
        plt.savefig('data_sample.png', dpi=150, bbox_inches='tight')
        plt.close()  # 明确关闭图形释放内存

    def _get_fallback_data(self):
        """生成符合KITTI标准尺寸的备用数据"""
        # KITTI标准尺寸
        w, h = 1242, 375

        # 生成网格点
        grid_x = np.linspace(0, w - 1, 15)
        grid_y = np.linspace(0, h - 1, 10)
        grid_pts = np.stack(np.meshgrid(grid_x, grid_y), axis=-1).reshape(-1, 2)
        pos = grid_pts.reshape(-1, 2)

        # 生成32维零描述符 (与模型输入维度匹配)
        descriptors = np.zeros((len(pos), 32), dtype=np.float32)  # 修复点：正确定义描述符

        # 强制生成合法边（连接相邻网格点）
        edges = []
        for i in range(len(pos)):
            if i % 15 != 14:  # 横向连接
                edges.append([i, i + 1])
            if i < len(pos) - 15:  # 纵向连接
                edges.append([i, i + 15])

        # 转换为张量并确保不越界
        edges = torch.tensor(edges, dtype=torch.long).t().contiguous()
        edges = edges[:, (edges[0] < len(pos)) & (edges[1] < len(pos))]  # 边界检查

        return Data(
            x=torch.tensor(descriptors, dtype=torch.float32),  # 使用定义好的描述符
            pos=torch.tensor(pos, dtype=torch.float32),
            edge_index=edges,
            num_nodes=len(pos),
            y=to_dense_adj(edges, max_num_nodes=len(pos)).squeeze(0)
        )

    def _load_calibration(self):
        """加载统一标定文件"""
        calib_path = os.path.join(self.root_dir, self.data_type, 'calib.txt')
        return parse_kitti_calib(calib_path)


class FeaturePyramid(nn.Module):
    def __init__(self):
        super().__init__()
        # 图像特征提取
        self.img_encoder = nn.Sequential(
            nn.Conv2d(3, 64, 3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(2),
            nn.Conv2d(64, 128, 3, padding=1),
            nn.ReLU()
        )

        # 点云特征提取
        self.pc_encoder = nn.Sequential(
            nn.Linear(3, 64),
            nn.ReLU(),
            nn.Linear(64, 128)
        )

        # 特征融合
        self.fusion = nn.Linear(256, 128)

    def forward(self, img, pointcloud):
        # 图像特征
        img_feat = self.img_encoder(img.permute(2, 0, 1).unsqueeze(0))
        img_feat = F.adaptive_avg_pool2d(img_feat, (1, 1)).squeeze()

        # 点云特征
        pc_feat = self.pc_encoder(pointcloud)
        pc_feat = torch.mean(pc_feat, dim=0)

        # 特征融合
        fused_feat = torch.cat([img_feat, pc_feat], dim=-1)
        return self.fusion(fused_feat)


class EnhancedFeaturePyramid(FeaturePyramid):
    def __init__(self):
        super().__init__()
        # 增加注意力机制
        self.attention = nn.MultiheadAttention(embed_dim=128, num_heads=4)

    def forward(self, img, pointcloud):
        base_features = super().forward(img, pointcloud)
        # 增加时序注意力
        attn_output, _ = self.attention(
            base_features.unsqueeze(1),
            base_features.unsqueeze(1),
            base_features.unsqueeze(1)
        )
        return base_features + attn_output.squeeze()


class TopologyBenchmark:
    METRIC_NAMES = ['Precision@0.5', 'H1_Consistency', 'Temporal_Stability']

    def __init__(self):
        self.results = {name: [] for name in self.METRIC_NAMES}

    def update(self, predictions, ground_truth):
        # 计算拓扑精度
        precision = self._calculate_precision(predictions, ground_truth)
        self.results['Precision@0.5'].append(precision)

        # 计算同调一致性
        h1_sim = self._calculate_h1_similarity(predictions, ground_truth)
        self.results['H1_Consistency'].append(h1_sim)

        # 计算时序稳定性
        if len(self.results['Temporal_Stability']) > 0:
            stability = self._calculate_stability(predictions, self.last_prediction)
            self.results['Temporal_Stability'].append(stability)
        self.last_prediction = predictions

    def _calculate_precision(self, pred, gt, threshold=0.5):
        pred_bin = (pred > threshold).astype(float)
        return np.sum(pred_bin * gt) / (np.sum(pred_bin) + 1e-7)

class AdaptiveParameters:
    def __init__(self):
        # 可调参数
        self.motion_thresh = 0.3
        self.risk_level = 1.0
        self.feature_quality = 0.8

    def update(self, frame_info):
        """基于场景复杂度动态调整参数"""
        # 计算场景复杂度
        gray = cv2.cvtColor(frame_info, cv2.COLOR_BGR2GRAY)
        entropy = -np.sum(gray * np.log2(gray + 1e-7))

        # 动态调整规则
        if entropy < 6:  # 简单场景
            self.motion_thresh = 0.2
            self.risk_level = 0.7
        elif 6 <= entropy < 8:  # 中等场景
            self.motion_thresh = 0.3
            self.risk_level = 1.0
        else:  # 复杂场景
            self.motion_thresh = 0.4
            self.risk_level = 1.5

        # 根据特征质量调整
        if self.feature_quality < 0.5:
            self.motion_thresh += 0.1

    for item in batch:

            valid_edges = (item.edge_index[0] <= max_node) & (item.edge_index[1] <= max_node)

def validate_graph(data):
    required = ['x', 'edge_index', 'colors', 'pos']
    if not all(hasattr(data, attr) for attr in required):
        return False
    if data.x.dim() != 2 or data.x.size(1) != 32:
        return False
    return True


def evaluate(model, dataloader, device, evaluator):
    model.eval()
    with torch.no_grad():
        for batch in dataloader:
            if batch is None:
                continue

            outputs = model(batch.to(device))

            # 将批次分解为单个图
            data_list = batch.to_data_list()

            # 分割预测结果
            edge_indices = []
            ptr = batch.ptr.cpu().numpy()  # 获取ptr数组
            edge_pred = outputs['edges'].cpu()

            # 遍历每个图的ptr区间
            for i in range(len(ptr) - 1):
                start_idx = ptr[i]
                end_idx = ptr[i + 1]

                # 提取属于当前图的边
                mask = (edge_pred[0] >= start_idx) & (edge_pred[0] < end_idx) & \
                       (edge_pred[1] >= start_idx) & (edge_pred[1] < end_idx)
                edges = edge_pred[:, mask]

                # 调整索引到当前图的局部范围
                edges -= start_idx
                edge_indices.append(edges)

            # 更新评估器
            for i, data in enumerate(data_list):
                pred = {
                    'vertices': data.pos,
                    'edges': edge_indices[i],
                    'weights': outputs['adj_matrix'][i].detach().cpu()  # 关键点
                }
                evaluator.update(pred, data)

    return evaluator.get_results()

class TopologyEvaluator:
    def __init__(self):
        self.reset()

    def reset(self):
        self.metrics = {
            'Precision@0.5': [],
            'H1_Consistency': [],
            'Temporal_Stability': []
        }
        self.last_pred = None

    def update(self, pred, gt):
        # 转换预测数据
        try:
            pred_edges = pred['edges'].cpu().numpy().T  # 转换为[2, E]格式
            pred_vertices = pred['vertices'].cpu().numpy()
        except AttributeError:
            pred_edges = np.array([])
            pred_vertices = np.array([])

        # 转换真实数据
        try:
            gt_edges = gt.edge_index.cpu().numpy().T  # 使用PyG的edge_index格式
            gt_vertices = gt.pos.cpu().numpy()
        except AttributeError:
            gt_edges = np.array([])
            gt_vertices = np.array([])

        # 计算精度（仅示例，需完善）
        precision = self._calc_precision(pred_edges, gt_edges)

        # 指标更新
        self.metrics['Precision@0.5'].append(precision)

        try:
            h1_sim = self._calc_h1_similarity(pred, gt)
            self.metrics['H1_Consistency'].append(h1_sim)
        except Exception as e:
            print(f"同调计算异常: {str(e)}")
            self.metrics['H1_Consistency'].append(0.0)

        # 时序稳定性计算（添加容错）
        if self.last_pred is not None:
            try:
                stability = self._calc_stability(pred, self.last_pred)
                self.metrics['Temporal_Stability'].append(stability)
            except Exception as e:
                print(f"稳定性计算异常: {str(e)}")
                self.metrics['Temporal_Stability'].append(0.0)

        # 更新历史记录（深拷贝避免引用问题）
        self.last_pred = {k: v.clone() if isinstance(v, torch.Tensor) else copy.deepcopy(v)
                          for k, v in pred.items()}
    def _calc_precision(self, pred_edges, gt_edges):
        """处理空输入和维度问题"""
        if len(pred_edges) == 0 or len(gt_edges) == 0:
            return 0.0

        # 确保转换为二维数组
        pred_edges = np.array(pred_edges).reshape(-1, 2)
        gt_edges = np.array(gt_edges).reshape(-1, 2)

        # 处理空数组
        if pred_edges.size == 0 or gt_edges.size == 0:
            return 0.0

        # 计算最大索引
        max_idx = max(
            pred_edges.max() if pred_edges.size > 0 else 0,
            gt_edges.max() if gt_edges.size > 0 else 0
        ) + 1

        # 创建邻接矩阵
        pred_adj = np.zeros((max_idx, max_idx))
        gt_adj = np.zeros((max_idx, max_idx))

        # 填充邻接矩阵 (添加边界检查)
        for u, v in pred_edges:
            if u < max_idx and v < max_idx:
                pred_adj[u, v] = 1
        for u, v in gt_edges:
            if u < max_idx and v < max_idx:
                gt_adj[u, v] = 1

        intersection = np.sum(pred_adj * gt_adj)
        union = np.sum(pred_adj)
        return intersection / (union + 1e-7)

    def _calc_h1_similarity(self, pred, gt):
        """计算H1同调相似度"""
        try:
            # 构建预测复形
            pred_complex = ChainComplex(
                vertices=pred['vertices'],
                edges=pred['edges'].t().tolist()
            )

            # 构建真实复形
            gt_complex = ChainComplex(
                vertices=gt['pos'].cpu().numpy(),
                edges=gt.edge_index.t().cpu().tolist()
            )

            # 计算同调
            pred_h1 = compute_homology(pred_complex)
            gt_h1 = compute_homology(gt_complex)

            # 计算相似度
            if len(gt_h1) == 0:
                return 0.0
            similarity = len(pred_h1) / len(gt_h1)
            return min(similarity, 1.0)  # 确保不超过1.0
        except Exception as e:
            print(f"H1相似度计算错误: {str(e)}")
            return 0.0

    def _calc_stability(self, current, previous):
        """计算时序稳定性指标"""
        try:
            current_adj = current['weights'].cpu().numpy()
            previous_adj = previous['weights'].cpu().numpy()
            return np.mean(np.abs(current_adj - previous_adj))
        except Exception as e:
            print(f"稳定性计算错误: {str(e)}")
            return 0.0

    def get_results(self):
        """返回包含详细序列和平均指标的字典"""
        return {
            'mean_metrics': {
                'Precision@0.5': np.nanmean(self.metrics['Precision@0.5']) if self.metrics['Precision@0.5'] else 0.0,
                'H1_Consistency': np.nanmean(self.metrics['H1_Consistency']) if self.metrics['H1_Consistency'] else 0.0,
                'Temporal_Stability': np.nanmean(self.metrics['Temporal_Stability']) if self.metrics[
                    'Temporal_Stability'] else 0.0
            },
            'full_sequence': {
                'Precision@0.5': self.metrics['Precision@0.5'],
                'H1_Consistency': self.metrics['H1_Consistency'],
                'Temporal_Stability': self.metrics['Temporal_Stability']
            }
        }

def calculate_curvature(points):
    """计算点集的近似曲率"""
    if len(points) < 3:
        return np.zeros(len(points))

    curvature = np.zeros(len(points))
    for i in range(1, len(points) - 1):
        # 使用三点差分法近似曲率
        x_prev, y_prev = points[i - 1]
        x_curr, y_curr = points[i]
        x_next, y_next = points[i + 1]

        dx1 = x_curr - x_prev
        dy1 = y_curr - y_prev
        dx2 = x_next - x_curr
        dy2 = y_next - y_curr

        # 计算角度变化
        angle1 = np.arctan2(dy1, dx1)
        angle2 = np.arctan2(dy2, dx2)
        delta_angle = angle2 - angle1

        # 曲率近似为角度变化率
        curvature[i] = delta_angle / np.sqrt(dx1 ** 2 + dy1 ** 2 + 1e-7)

    # 处理边界点
    curvature[0] = curvature[1]
    curvature[-1] = curvature[-2]
    return curvature



def delaunay_triangulation(points, frame_size):
    edge_tensor = torch.zeros((2, 0), dtype=torch.long)
    if len(points) == 0:
        return edge_tensor

    # 初始化列表收集所有点
    filtered_points_list = []
    valid_indices = []
    valid_edges = set()

    # 第一次收集有效点
    for idx, p in enumerate(points):
        if 0 <= p[0] < frame_size[0] and 0 <= p[1] < frame_size[1]:
            valid_indices.append(idx)
            filtered_points_list.append(p)

    # 添加网格点（转换为列表）
    grid_x = np.linspace(0, frame_size[0], 15)
    grid_y = np.linspace(0, frame_size[1], 10)
    grid_points = np.array(np.meshgrid(grid_x, grid_y)).T.reshape(-1, 2)
    filtered_points_list.extend(grid_points.tolist())

    # 添加边界点
    w, h = frame_size
    border_points = [(0, 0), (w - 1, 0), (w - 1, h - 1), (0, h - 1)]
    filtered_points_list.extend(border_points)

    # 转换为NumPy数组
    filtered_points = np.array(filtered_points_list)

    # 后续处理（使用NumPy数组，不再使用append）
    if len(filtered_points) > 2:
        curvature = calculate_curvature(filtered_points)
        for i in range(len(curvature) - 1):
            if abs(curvature[i]) > 0.1:
                valid_edges.add((i, i + 1))
    # 空间均匀采样

    # 添加网格关键点（保持原逻辑）
    if len(filtered_points) < 20:
        grid_x = np.linspace(0, frame_size[0], 10)
        grid_y = np.linspace(0, frame_size[1], 10)
        grid_points = np.array(np.meshgrid(grid_x, grid_y)).T.reshape(-1, 2)
        filtered_points.extend(grid_points)

    # 边界处理（保持原逻辑）
    w, h = frame_size
    border_points = [(0, 0), (w - 1, 0), (w - 1, h - 1), (0, h - 1)]
    filtered_points = np.vstack([filtered_points, border_points])
    points = np.clip(points, [0, 0], [frame_size[0] - 1e-5, frame_size[1] - 1e-5])
    # 增加最小点集检查
    if len(points) < 3:
        return torch.zeros((2, 0), dtype=torch.long)
    try:
        # Delaunay剖分主逻辑v
        subdiv = cv2.Subdiv2D((0, 0, w, h))
        seen = set()
        for p in filtered_points:
            pt = (max(0, min(p[0], w - 1)), max(0, min(p[1], h - 1)))
            if pt not in seen:
                subdiv.insert(pt)
                seen.add(pt)

        # 提取三角形边
        triangles = subdiv.getTriangleList()
        valid_edges = set()

        for t in triangles:
            pts = [(t[0], t[1]), (t[2], t[3]), (t[4], t[5])]
            indices = []
            for p in pts:
                for idx, fp in enumerate(filtered_points):
                    if np.linalg.norm(fp - p) < 1e-3:
                        indices.append(idx)
                        break
            if len(indices) == 3:
                valid_edges.update([(indices[0], indices[1]),
                                    (indices[1], indices[2]),
                                    (indices[2], indices[0])])

        # 曲率增强处理
        if len(filtered_points) > 2:
            curvature = calculate_curvature(filtered_points)
            for i in range(len(curvature) - 1):
                if abs(curvature[i] - curvature[i + 1]) > 0.1:
                    valid_edges.add((i, i + 1))

        # 转换为张量

        valid_edges = [
            (u, v) for u, v in valid_edges
            if u < len(filtered_points) and v < len(filtered_points)
        ]

        # 转换为张量
        if valid_edges:
            edge_tensor = torch.tensor(list(valid_edges), dtype=torch.long).t().contiguous()
        else:
            edge_tensor = torch.zeros((2, 0), dtype=torch.long)

    except Exception as e:
        print(f"Delaunay剖分异常: {str(e)}")

    return edge_tensor

def load_checkpoint(model, checkpoint_path, device):
    try:
        checkpoint = torch.load(checkpoint_path, map_location=device)
        model_dict = model.state_dict()

        # 智能参数匹配
        pretrained_dict = {}
        for k, v in checkpoint.items():
            # 处理不同参数名称
            if k not in model_dict:
                possible_keys = [
                    k.replace('conv1', 'gcn1'),
                    k.replace('edge_weight.0', 'edge_weight.3')
                ]
                for pk in possible_keys:
                    if pk in model_dict:
                        pretrained_dict[pk] = v
                        break
            else:
                # 处理尺寸不匹配
                if v.shape != model_dict[k].shape:
                    if 'weight' in k and len(v.shape) == 2:
                        min_dim = min(v.shape[0], model_dict[k].shape[0])
                        pretrained_dict[k] = model_dict[k]
                        pretrained_dict[k][:min_dim] = v[:min_dim]
                    else:
                        print(f"跳过参数 {k} (形状不匹配)")
                else:
                    pretrained_dict[k] = v

        model.load_state_dict(pretrained_dict, strict=False)
        print(f"成功加载 {len(pretrained_dict)}/{len(model_dict)} 参数")
        return True
    except Exception as e:
        print(f"加载检查点失败: {str(e)}")
        return False

def train_model(config, args):
    best_loss = float('inf')
    os.makedirs("checkpoints", exist_ok=True)

    # 初始化数据集
    train_dataset = KITTIDataset(config['data_root'], split='train')
    val_dataset = KITTIDataset(config['data_root'], split='val')

    # 数据加载器配置
    train_loader = DataLoader(
        train_dataset,
        batch_size=config['batch_size'],
        shuffle=True,
        collate_fn=collate_graphs,
        num_workers=0,
        pin_memory=True  # 提升GPU传输效率
    )

    val_loader = DataLoader(
        val_dataset,
        batch_size=config['batch_size'],
        collate_fn=collate_graphs,
        num_workers=0
    )

    # 模型配置
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    model = TopologyRepair(in_channels=32).to(device)

    # 优化器配置（调整学习率）
    optimizer = torch.optim.AdamW(model.parameters(), lr=3e-4, weight_decay=1e-4)
    scheduler = torch.optim.lr_scheduler.ReduceLROnPlateau(
        optimizer,
        mode='min',
        factor=0.5,
        patience=3,
        verbose=True
    )

    # 训练循环
    for epoch in range(config['epochs']):
        print(f"\n=== Epoch {epoch + 1}/{config['epochs']} ===")

        # 训练阶段
        model.train()
        epoch_train_loss = 0.0
        processed_batches = 0

        for batch_idx, batch in enumerate(train_loader):
            try:
                # 跳过无效批次
                if batch is None or batch.num_nodes < 2:
                    continue

                # 设备转移
                batch = batch.to(device)

                # 梯度清零
                optimizer.zero_grad(set_to_none=True)  # 更高效的清零方式

                # 前向传播
                output = model(batch)
                loss = output['loss']

                # 反向传播
                loss.backward()

                # 梯度裁剪
                torch.nn.utils.clip_grad_norm_(
                    model.parameters(),
                    max_norm=5.0,
                    error_if_nonfinite=True
                )

                # 参数更新
                optimizer.step()

                # 记录损失
                epoch_train_loss += loss.item()
                processed_batches += 1

            except RuntimeError as e:
                if 'CUDA out of memory' in str(e):
                    print("检测到显存不足，跳过当前批次")
                    torch.cuda.empty_cache()
                    continue
                else:
                    raise e

            # 验证阶段
        model.eval()
        val_loss = 0.0
        with torch.no_grad(), torch.cuda.amp.autocast():
            for val_batch in val_loader:
                if val_batch is None:
                    continue
                val_batch = val_batch.to(device)
                output = model(val_batch)
                val_loss += output['loss'].item()

        # 计算平均损失
        avg_train_loss = epoch_train_loss / processed_batches if processed_batches > 0 else float('nan')
        avg_val_loss = val_loss / len(val_loader) if len(val_loader) > 0 else float('nan')
        # 保存最佳模型
        if avg_val_loss < best_loss:
            best_loss = avg_val_loss
            torch.save(
                model.state_dict(),
                f"checkpoints/best_epoch_{epoch + 1}.pth"
            )
            print(f"保存最佳模型（Loss: {best_loss:.4f}）")

        # 打印epoch总结
        print(f"\nEpoch {epoch + 1} 总结:")
        print(f"训练损失: {avg_train_loss:.4f}")
        print(f"验证损失: {avg_val_loss:.4f}")
        print(f"当前学习率: {optimizer.param_groups[0]['lr']:.2e}")

    print("训练完成！")
